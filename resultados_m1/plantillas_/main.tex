\newcommand{\bubbleletter}[1]{%
\tikz[baseline=-0.6ex]{ 
    \draw[line width=0.6pt, color=gristexto] (0,0) circle (0.16cm);
    \node[font=\tiny\bfseries, text=gristexto] at (0,0) {#1};
}}

    \node[font=\fontsize{5}{5}\selectfont, text=azulpreu] at (0,0) {#1};
}}
